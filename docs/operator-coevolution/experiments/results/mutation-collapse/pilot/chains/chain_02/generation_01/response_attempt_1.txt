{Replaces deterministic block-level optimization with stochastic sequential greedy insertion. Customers are reordered randomly and inserted individually into the first feasible candidate position minimizing cost, verifying feasibility before commitment.}

def repair_operator(problem, routes, removed, rng):
    result = clone_routes(routes)
    if not removed:
        return result
    
    # Create a local copy of removed to avoid mutating the input argument
    work_order = list(removed)
    # Introduce stochasticity to escape local optima traps associated with fixed block orders
    rng.shuffle(work_order)
    
    for customer in work_order:
        candidates = insertion_candidates(problem, result, customer)
        
        best_feasible = None
        # Iterate through cost-sorted candidates to find the first feasible one
        for delta, route_idx, pos in candidates:
            # Validate feasibility without mutating result
            current_route = result[route_idx]
            # Construct temporary route for feasibility check
            temp_route = current_route[:pos] + [customer] + current_route[pos:]
            
            if route_feasible(problem, temp_route):
                best_feasible = (delta, route_idx, pos)
                break
        
        if best_feasible:
            apply_insertion(result, customer, best_feasible)
        elif candidates:
            # Fallback to parent's blind insertion strategy if no feasible candidate found
            apply_insertion(result, customer, candidates[0])
            
    return result
